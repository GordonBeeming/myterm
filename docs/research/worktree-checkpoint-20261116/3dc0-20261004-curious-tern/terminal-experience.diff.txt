+        func waitFor(_ condition: () -> Bool) async throws {
+            for _ in 0..<200 {
+                if condition() { return }

+                                      sessionID: state.route.sessionID),

+                sequence: 0, chunkIndex: 0, chunkCount: 1, totalBytes: bytes.count, bytes: bytes))

+        XCTAssertFalse(state.isAwaitingCheckpoint)
